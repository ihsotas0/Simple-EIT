import os
import joblib
import hashlib
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from datetime import datetime

from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
from sklearn.calibration import CalibratedClassifierCV

from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.linear_model import LogisticRegression
from sklearn.neighbors import KNeighborsClassifier
from sklearn.neural_network import MLPClassifier
from sklearn.svm import SVC

from xgboost import XGBClassifier

class Classifier:

    def __init__(self):
        self.model_name = None
        self.object_name = None

        self.model = None
        self.scaler = StandardScaler()
        self.encoder = LabelEncoder()

        self.X_train, self.X_test = None, None
        self.y_train, self.y_test = None, None

        self.loss_curve = None

        self.cache_dir = "../data/models/"
        os.makedirs(self.cache_dir, exist_ok=True)

        self.dataset_map = {
            'vertical_eraser': '../data/archive/vertical_eraser_no_salt_data_formatted.csv',
        }

        self.model_map = {
            'gb': lambda: GradientBoostingClassifier(),
            'knn': lambda: KNeighborsClassifier(n_neighbors=6),
            'logreg': lambda: LogisticRegression(max_iter=2000),
            'mlp': lambda: MLPClassifier(hidden_layer_sizes=(10, 10), max_iter=10000, random_state=1),
            'lda': lambda: LinearDiscriminantAnalysis(),
            'rf': lambda: RandomForestClassifier(n_estimators=200, random_state=42),
            'svm': lambda: SVC(kernel='rbf', probability=True),
            'xgb': lambda: XGBClassifier(n_estimators=300, learning_rate=0.05, max_depth=4, verbosity=0),
        }

    # -------------------------
    # Utilities
    # -------------------------
    def list_models(self):
        return list(self.model_map.keys())

    def list_objects(self):
        return list(self.dataset_map.keys())

    def _hash_dataset(self, df):
        return hashlib.md5(pd.util.hash_pandas_object(df, index=True).values).hexdigest()

    def _cache_path(self, dataset_hash):
        filename = f"{self.object_name}_{self.model_name}_{dataset_hash}.joblib"
        return os.path.join(self.cache_dir, filename)

    # -------------------------
    # Dataset loading
    # -------------------------
    def set_object(self, object_name):
        if object_name not in self.dataset_map:
            raise ValueError(f"Dataset '{object_name}' not found.")
        self.object_name = object_name
        df = pd.read_csv(self.dataset_map[object_name])
        self.dataset_hash = self._hash_dataset(df)

        X = df.drop("Label", axis=1).values
        y = df["Label"].values
        self.encoder.fit(y)
        X_scaled = self.scaler.fit_transform(X)

        self.X_train, self.X_test, self.y_train, self.y_test = train_test_split(
            X_scaled, self.encoder.transform(y), test_size=0.2, random_state=1
        )

    # -------------------------
    # Model selection
    # -------------------------
    def set_model(self, model_name, calibrate=True):
        if model_name not in self.model_map:
            raise ValueError(f"Model '{model_name}' not supported.")
        self.model_name = model_name
        model = self.model_map[model_name]()
        # Wrap with probability calibration if supported
        if calibrate and hasattr(model, "predict_proba"):
            self.model = CalibratedClassifierCV(model, cv="prefit") if hasattr(model, "fit") else model
        else:
            self.model = model

    # -------------------------
    # Train/load with caching
    # -------------------------
    def fit(self):
        if self.model is None:
            raise ValueError("No model selected.")
        cache_path = self._cache_path(self.dataset_hash)
        if os.path.exists(cache_path):
            print(f"Loading cached model: {cache_path}")
            data = joblib.load(cache_path)
            self.model = data["model"]
            self.scaler = data["scaler"]
            self.encoder = data["encoder"]
            self.loss_curve = data.get("loss_curve", None)
            return

        print("Training model...")
        self.model.fit(self.X_train, self.y_train)

        if hasattr(self.model, "loss_curve_"):
            self.loss_curve = self.model.loss_curve_

        acc = self.evaluate()
        joblib.dump({
            "model": self.model,
            "scaler": self.scaler,
            "encoder": self.encoder,
            "loss_curve": self.loss_curve,
            "metadata": {"accuracy": float(acc), "timestamp": str(datetime.now())}
        }, cache_path)
        print(f"Model saved to {cache_path}. Accuracy: {acc:.3f}")

    # -------------------------
    # Predict
    # -------------------------
    def predict(self, v):
        v = np.array(v).reshape(1, -1)
        v_scaled = self.scaler.transform(v)
        if hasattr(self.model, "predict_proba"):
            probs = self.model.predict_proba(v_scaled)[0]
        else:
            pred = self.model.predict(v_scaled)[0]
            probs = np.zeros(len(self.encoder.classes_))
            probs[pred] = 1.0
        return self.format_output(probs)

    # -------------------------
    # Format output
    # -------------------------
    def format_output(self, probs):
        label_order = self.encoder.classes_
        prob_dict = {label: prob for label, prob in zip(label_order, probs)}
        row1 = [prob_dict.get(label, 0) for label in ["AB_1","AB_2","AD_1","AD_2","CD_1","CD_2","BC_1","BC_2"]]
        row2 = [prob_dict.get(label, 0) for label in ["AB_3","AB_4","AD_3","AD_4","CD_3","CD_4","BC_3","BC_4"]]
        return np.array([row1,row2])

    # -------------------------
    # Evaluation
    # -------------------------
    def evaluate(self):
        preds = self.model.predict(self.X_test)
        return accuracy_score(self.y_test, preds)

    # -------------------------
    # Plot loss
    # -------------------------
    def plot_loss(self):
        if self.loss_curve is None:
            print("No loss curve available")
            return
        plt.figure(figsize=(6,4))
        plt.plot(self.loss_curve)
        plt.title(f"{self.model_name.upper()} Loss vs Epochs")
        plt.xlabel("Epochs")
        plt.ylabel("Loss")
        plt.grid(True)
        plt.show()

    # -------------------------
    # Model leaderboard
    # -------------------------
    def compare_models(self):
        results = {}
        for name in self.model_map.keys():
            self.set_model(name)
            self.fit()
            acc = self.evaluate()
            results[name] = acc
        print("\nModel Leaderboard:")
        for m, a in sorted(results.items(), key=lambda x: x[1], reverse=True):
            print(f"{m}: {a:.3f}")
        return results
